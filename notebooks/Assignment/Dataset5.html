import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

hours_bins = [2, 4, 6, 8, np.inf]
hours_labels = ["2-4h", "4-6h", "6-8h", "8h+"]
df["sleep_band"] = pd.cut(df["sleep_hours"].clip(lower=0),
 
  bins=hours_bins, labels=hours_labels, right=False, include_lowest=True)

counts = df["sleep_band"].value_counts().sort_index()
plt.figure(figsize=(6,6))
plt.pie(counts, labels=counts.index.astype(str), autopct="%1.1f%%", startangle=140)
plt.title("Sleep hour distribution")
plt.show()

This plot box also shows that most students sleep between 6 and 8 hours. The average is around 7 hours.
